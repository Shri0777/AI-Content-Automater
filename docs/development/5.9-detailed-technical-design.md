```python
def can_transition_to(self, new_state):
    valid_transitions = {
        'draft': ['in_progress'],
        'in_progress': ['pending_review', 'draft'],
        'pending_review': ['approved', 'draft'],
        'approved': ['published'],
        'published': ['archived'],
        'archived': []
    }
    return new_state in valid_transitions.get(self.lifecycle_state, [])

def transition_to(self, new_state):
    if not self.can_transition_to(new_state):
        raise ValidationError(f"Cannot transition from {self.lifecycle_state} to {new_state}")
    self.lifecycle_state = new_state
    self.save(update_fields=['lifecycle_state', 'updated_at'])
```

```python
class Migration(migrations.Migration):
```

```python
            )
```

```python
# content/services/research_service.py

import feedparser
import requests
from datetime import datetime, timedelta
from django.utils import timezone
from urllib.parse import quote_plus

class RSSFeedService:
    """
    Service for parsing RSS/Atom feeds.
    V3 Requirement: Free research source.
    """
    
    def fetch_feed(self, feed_url, max_entries=20):
        """
        Fetch and parse RSS/Atom feed.
        
        Returns: List of dicts with standardized fields:
        {
            'title': str,
            'summary': str,
            'url': str,
            'source_name': str,
            'author': str,
            'published_date': datetime,
        }
        """
        try:
            feed = feedparser.parse(feed_url)
            
            if feed.bozo:  # Parse error
                return {'error': f'Feed parse error: {feed.bozo_exception}'}
            
            entries = []
            source_name = feed.feed.get('title', 'Unknown Source')
            
            for entry in feed.entries[:max_entries]:
                entries.append({
                    'title': entry.get('title', 'Untitled'),
                    'summary': entry.get('summary', entry.get('description', '')),
                    'url': entry.get('link', ''),
                    'source_name': source_name,
                    'author': entry.get('author', ''),
                    'published_date': self._parse_date(entry.get('published', entry.get('updated'))),
                })
            
            return {'entries': entries, 'feed_title': source_name}
            
        except Exception as e:
            return {'error': str(e)}
    
    def _parse_date(self, date_string):
        """Parse RSS date to datetime."""
        if not date_string:
            return None
        
        # feedparser provides struct_time in published_parsed
        # Convert to datetime
        try:
            from time import mktime
            return datetime.fromtimestamp(mktime(feedparser._parse_date(date_string)))
        except:
            return None


class GNewsService:
    """
    Service for querying Google News RSS.
    V3 Requirement: Free news research source.
    
    Google News RSS URL pattern:
    https://news.google.com/rss/search?q={query}&hl={language}&gl={country}&ceid={country}:{language}
    """
    
    BASE_URL = "https://news.google.com/rss/search"
    
    def search(self, query, language='en', country='US', max_results=20, days_back=7):
        """
        Search Google News via RSS.
        
        Args:
            query: Search terms
            language: Language code (en, es, fr, etc.)
            country: Country code (US, GB, IN, etc.)
            max_results: Max articles to return
            days_back: Only articles from last N days
        
        Returns: Same structure as RSSFeedService.fetch_feed()
        """
        try:
            # Build query with date filter
            if days_back:
                date_filter = f" when:{days_back}d"
                query_with_date = query + date_filter
            else:
                query_with_date = query
            
            # Build URL
            params = {
                'q': query_with_date,
                'hl': language,
                'gl': country,
                'ceid': f"{country}:{language}"
            }
            
            url = f"{self.BASE_URL}?q={quote_plus(params['q'])}&hl={params['hl']}&gl={params['gl']}&ceid={params['ceid']}"
            
            # Use RSS service to parse
            rss_service = RSSFeedService()
            result = rss_service.fetch_feed(url, max_entries=max_results)
            
            # Override source_name for GNews entries
            if 'entries' in result:
                for entry in result['entries']:
                    entry['source_name'] = 'Google News'
            
            return result
            
        except Exception as e:
            return {'error': str(e)}


class ResearchCollectionService:
    """
    Orchestrates research collection from multiple sources.
    V3 Requirements: Quality, deduplication, provenance.
    """
    
    def __init__(self):
        self.rss_service = RSSFeedService()
        self.gnews_service = GNewsService()
    
    def collect_research(self, creation_request):
        """
        Collect research for a ContentCreationRequest.
        
        V3 Requirements enforced:
        - Freshness filtering
        - Source category filtering
        - Deduplication by URL
        - Relevance scoring
        - Partial results (preserve what works, report failures)
        """
        results = []
        errors = []
        
        # Extract parameters from request
        topics = creation_request.topics
        freshness_days = creation_request.freshness_days
        categories = creation_request.source_categories
        language = creation_request.language
        
        # Collect from GNews if enabled
        if 'news' in categories:
            for topic in topics:
                gnews_result = self.gnews_service.search(
                    query=topic,
                    language=language,
                    days_back=freshness_days,
                    max_results=10
                )
                
                if 'error' in gnews_result:
                    errors.append(f"GNews ({topic}): {gnews_result['error']}")
                elif 'entries' in gnews_result:
                    results.extend(gnews_result['entries'])
        
        # Collect from RSS feeds if enabled
        if any(cat in categories for cat in ['blogs', 'documentation']):
            # Get user's RSS feeds or use defaults
            feeds = self._get_rss_feeds(creation_request.user, categories)
            
            for feed_url in feeds:
                rss_result = self.rss_service.fetch_feed(feed_url, max_entries=10)
                
                if 'error' in rss_result:
                    errors.append(f"RSS ({feed_url}): {rss_result['error']}")
                elif 'entries' in rss_result:
                    results.extend(rss_result['entries'])
        
        # Deduplicate by URL
        seen_urls = set()
        deduplicated = []
        for item in results:
            if item['url'] and item['url'] not in seen_urls:
                seen_urls.add(item['url'])
                deduplicated.append(item)
        
        # Filter by freshness if published_date available
        if creation_request.allow_older_sources:
            # Keep all
            filtered = deduplicated
        else:
            # Only fresh items
            cutoff = timezone.now() - timedelta(days=freshness_days)
            filtered = [
                item for item in deduplicated
                if not item['published_date'] or item['published_date'] >= cutoff
            ]
        
        # Calculate relevance scores
        for item in filtered:
            item['relevance_score'] = self._calculate_relevance(item, topics)
        
        # Sort by relevance
        filtered.sort(key=lambda x: x['relevance_score'], reverse=True)
        
        return {
            'results': filtered,
            'errors': errors,
            'total_collected': len(results),
            'after_deduplication': len(deduplicated),
            'after_filtering': len(filtered)
        }
    
    def _get_rss_feeds(self, user, categories):
        """
        Get RSS feeds for user and categories.
        TODO: Implement RSSFeedSource model for user-managed feeds.
        For now, return default feeds.
        """
        defaults = {
            'blogs': [
                'https://martinfowler.com/feed.atom',
                'https://blog.cloudflare.com/rss/',
                'https://aws.amazon.com/blogs/aws/feed/',
            ],
            'documentation': [
                'https://kubernetes.io/feed.xml',
                'https://www.docker.com/blog/feed/',
            ]
        }
        
        feeds = []
        for category in categories:
            feeds.extend(defaults.get(category, []))
        
        return feeds
    
    def _calculate_relevance(self, item, topics):
        """
        Simple keyword-based relevance scoring.
        V3 Requirement: Relevance to user's topics.
        
        Returns: 0-1 score
        """
        text = f"{item['title']} {item['summary']}".lower()
        
        matches = 0
        for topic in topics:
            if topic.lower() in text:
                matches += 1
        
        if not topics:
            return 0.5  # Neutral if no topics
        
        return min(matches / len(topics), 1.0)
```

```python
    V3 Requirements: RSS/GNews collection, quality display, provenance.
    
    URL: /content/create/<request_id>/research/
    Template: content/content_research_collect.html
    
    Flow:
    1. Load ContentCreationRequest (ownership check)
    2. If status='configuring', trigger collection
    3. Display collected research with quality metrics
    4. Allow manual editing/filtering
    5. Proceed to content generation
    """
    creation_request = get_object_or_404(
        ContentCreationRequest,
        pk=request_id,
        user=request.user
    )
    
    # Trigger collection on first visit
    if creation_request.status == 'configuring':
        service = ResearchCollectionService()
        collection_result = service.collect_research(creation_request)
        
        # Save research items
        for item in collection_result['results']:
            Research.objects.create(
                owner=request.user,
                creation_request=creation_request,
                entry_type='gnews' if item['source_name'] == 'Google News' else 'rss',
                title=item['title'],
                summary=item['summary'],
                url=item['url'],
                source_name=item['source_name'],
                author=item.get('author', ''),
                published_date=item.get('published_date'),
                relevance_score=item['relevance_score'],
                category=item.get('category', ''),
            )
        
        # Update request status
        creation_request.status = 'researching'
        creation_request.save()
        
        messages.success(
            request,
            f"Collected {collection_result['after_filtering']} research items. "
            f"({collection_result['total_collected']} total, "
            f"{collection_result['after_deduplication']} after deduplication)"
        )
        
        if collection_result['errors']:
            errors_preview = ', '.join(collection_result['errors'][:3])
            if len(collection_result['errors']) > 3:
                errors_preview += f" (and {len(collection_result['errors']) - 3} more)"
            messages.warning(request, f"Some sources failed: {errors_preview}")
    
    # Display collected research
    research_items = creation_request.research_results.all().order_by(
        '-relevance_score',
        '-published_date'
    )
    
    # Calculate quality statistics
    total_items = research_items.count()
    fresh_items = sum(1 for item in research_items if item.is_fresh(creation_request.freshness_days))
    avg_relevance = research_items.aggregate(models.Avg('relevance_score'))['relevance_score__avg'] or 0
    
    return render(request, 'content/content_research_collect.html', {
        'creation_request': creation_request,
        'research_items': research_items,
        'stats': {
            'total': total_items,
            'fresh': fresh_items,
            'avg_relevance': round(avg_relevance, 2),
```

```python
# content/authorization.py

def user_owns_research(view_func):
    """
    Decorator ensuring user owns the research item.
    Similar to @user_owns_content.
    """
    @wraps(view_func)
    def wrapper(request, *args, **kwargs):
        research_pk = kwargs.get('pk') or kwargs.get('research_pk')
        research = get_object_or_404(Research, pk=research_pk)
        
        if research.owner != request.user:
            raise PermissionDenied("You do not own this research item.")
        
        return view_func(request, *args, **kwargs)
    return wrapper
```

**Apply to:**
- Research detail view
- Research edit view
- Research delete view
- Research-to-content association view
